# Project: Buku Nilai Siswa

Sistem manajemen nilai siswa yang menggunakan list dan dictionary!

## Struktur Data

```python
# Setiap siswa disimpan sebagai dictionary
# Semua siswa dalam sebuah list
siswa_list = []

def buat_siswa(nama, nilai_list):
    return {
        "nama": nama,
        "nilai": nilai_list,
        "rata_rata": sum(nilai_list) / len(nilai_list) if nilai_list else 0
    }
```

## Program Lengkap

```python
def hitung_grade(rata):
    if rata >= 90: return "A"
    elif rata >= 80: return "B"
    elif rata >= 70: return "C"
    elif rata >= 60: return "D"
    else: return "E"

def tambah_siswa(data, nama, nilai):
    rata = sum(nilai) / len(nilai) if nilai else 0
    data.append({"nama": nama, "nilai": nilai, "rata": rata})
    print(f"✅ {nama} berhasil ditambahkan.")

def tampilkan_semua(data):
    if not data:
        print("Belum ada data siswa.")
        return
    
    print("\n" + "=" * 55)
    print(f"{'No':>3} {'Nama':<15} {'Nilai':^20} {'Rata':>6} {'Grade':>5}")
    print("=" * 55)
    
    for i, s in enumerate(data, 1):
        nilai_str = str(s["nilai"]).replace("[", "").replace("]", "")
        print(f"{i:>3} {s['nama']:<15} {nilai_str:^20} {s['rata']:>6.1f} {hitung_grade(s['rata']):>5}")
    
    print("=" * 55)

def cari_siswa(data, nama):
    for s in data:
        if s["nama"].lower() == nama.lower():
            return s
    return None

def statistik_kelas(data):
    if not data:
        print("Belum ada data.")
        return
    
    semua_rata = [s["rata"] for s in data]
    print("\n=== Statistik Kelas ===")
    print(f"Jumlah siswa: {len(data)}")
    print(f"Rata-rata kelas: {sum(semua_rata)/len(semua_rata):.1f}")
    print(f"Nilai tertinggi: {max(semua_rata):.1f}")
    print(f"Nilai terendah: {min(semua_rata):.1f}")
    
    lulus = sum(1 for r in semua_rata if r >= 60)
    print(f"Siswa lulus: {lulus}/{len(data)}")

def main():
    data_siswa = []
    
    # Data awal
    tambah_siswa(data_siswa, "Budi", [85, 92, 88])
    tambah_siswa(data_siswa, "Siti", [92, 95, 90])
    tambah_siswa(data_siswa, "Andi", [70, 75, 68])
    tambah_siswa(data_siswa, "Rina", [88, 85, 90])
    tambah_siswa(data_siswa, "Dodi", [55, 60, 58])
    
    while True:
        print("\n╔═══════════════════════╗")
        print("║   BUKU NILAI SISWA    ║")
        print("╠═══════════════════════╣")
        print("║ 1. Tampilkan semua    ║")
        print("║ 2. Tambah siswa       ║")
        print("║ 3. Cari siswa         ║")
        print("║ 4. Statistik kelas    ║")
        print("║ 5. Keluar             ║")
        print("╚═══════════════════════╝")
        
        pilihan = input("\nPilihan: ").strip()
        
        if pilihan == "1":
            tampilkan_semua(data_siswa)
        
        elif pilihan == "2":
            nama = input("Nama siswa: ").strip()
            nilai_input = input("Masukkan nilai (pisah koma): ")
            nilai = [float(n.strip()) for n in nilai_input.split(",")]
            tambah_siswa(data_siswa, nama, nilai)
        
        elif pilihan == "3":
            nama = input("Cari nama: ")
            siswa = cari_siswa(data_siswa, nama)
            if siswa:
                print(f"\nDitemukan: {siswa['nama']}")
                print(f"Nilai: {siswa['nilai']}")
                print(f"Rata-rata: {siswa['rata']:.1f}")
                print(f"Grade: {hitung_grade(siswa['rata'])}")
            else:
                print(f"Siswa '{nama}' tidak ditemukan.")
        
        elif pilihan == "4":
            statistik_kelas(data_siswa)
        
        elif pilihan == "5":
            print("Sampai jumpa!")
            break
        
        else:
            print("Pilihan tidak valid!")

main()
```
