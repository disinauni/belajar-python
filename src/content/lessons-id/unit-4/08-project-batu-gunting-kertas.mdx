# Project: Batu Gunting Kertas

Game klasik yang akan kita buat dengan fungsi-fungsi yang terorganisir!

## Aturan Game

- Batu mengalahkan Gunting
- Gunting mengalahkan Kertas
- Kertas mengalahkan Batu
- Sama → Seri

## Implementasi

```python
import random

def dapatkan_pilihan_komputer():
    """Pilih secara acak untuk komputer."""
    pilihan = ["batu", "gunting", "kertas"]
    return random.choice(pilihan)

def tentukan_pemenang(pemain, komputer):
    """Tentukan pemenang ronde ini."""
    if pemain == komputer:
        return "seri"
    
    menang = {
        "batu": "gunting",
        "gunting": "kertas",
        "kertas": "batu"
    }
    
    if menang[pemain] == komputer:
        return "pemain"
    else:
        return "komputer"

def tampilkan_hasil(pemain, komputer, hasil):
    """Tampilkan hasil ronde dengan emoji."""
    emoji = {"batu": "🪨", "gunting": "✂️", "kertas": "📄"}
    
    print(f"\nKamu    : {emoji[pemain]} {pemain.capitalize()}")
    print(f"Komputer: {emoji[komputer]} {komputer.capitalize()}")
    
    if hasil == "seri":
        print("🤝 SERI!")
    elif hasil == "pemain":
        print("🎉 KAMU MENANG!")
    else:
        print("💻 KOMPUTER MENANG!")

def minta_pilihan_pemain():
    """Minta input dari pemain dengan validasi."""
    pilihan_valid = ["batu", "gunting", "kertas"]
    
    while True:
        print("\nPilihan: batu | gunting | kertas | keluar")
        masukan = input("Pilihanmu: ").lower().strip()
        
        if masukan == "keluar":
            return None
        if masukan in pilihan_valid:
            return masukan
        print("Pilihan tidak valid! Coba lagi.")

def main():
    """Fungsi utama — jalankan game."""
    print("╔══════════════════════════════╗")
    print("║   BATU GUNTING KERTAS 🎮     ║")
    print("╚══════════════════════════════╝")
    
    nama = input("\nNamamu: ")
    print(f"\nHalo {nama}! Siap bermain?")
    
    skor_pemain = 0
    skor_komputer = 0
    ronde = 0
    
    while True:
        print(f"\n--- Ronde {ronde + 1} | Skor: {nama} {skor_pemain} - {skor_komputer} Komputer ---")
        
        pilihan_pemain = minta_pilihan_pemain()
        
        if pilihan_pemain is None:
            print("\nKamu memilih keluar.")
            break
        
        pilihan_komputer = dapatkan_pilihan_komputer()
        hasil = tentukan_pemenang(pilihan_pemain, pilihan_komputer)
        
        tampilkan_hasil(pilihan_pemain, pilihan_komputer, hasil)
        
        ronde += 1
        if hasil == "pemain":
            skor_pemain += 1
        elif hasil == "komputer":
            skor_komputer += 1
    
    # Rekap akhir
    print(f"\n{'='*35}")
    print(f"HASIL AKHIR setelah {ronde} ronde:")
    print(f"{nama}: {skor_pemain} menang")
    print(f"Komputer: {skor_komputer} menang")
    seri = ronde - skor_pemain - skor_komputer
    print(f"Seri: {seri}")
    
    if skor_pemain > skor_komputer:
        print(f"\n🏆 {nama} MEMENANGKAN PERTANDINGAN!")
    elif skor_komputer > skor_pemain:
        print("\n🤖 Komputer menang pertandingan ini.")
    else:
        print("\n🤝 Pertandingan berakhir seri!")

# Jalankan game
main()
```

## Apa yang Sudah Dikuasai?

| Konsep | Digunakan di |
|--------|-------------|
| `def` | Semua fungsi |
| Parameter & argumen | `tentukan_pemenang(pemain, komputer)` |
| Return values | Mengembalikan hasil/pilihan |
| Default args | (bisa ditambahkan) |
| Scope | Variabel skor di `main()` |
