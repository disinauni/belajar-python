# Final Project: Manajemen Kontak

Ini adalah proyek final! Kamu akan membuat **sistem manajemen kontak** yang menggabungkan semua yang sudah dipelajari.

## Fitur Aplikasi

- ✅ Tambah kontak baru
- ✅ Lihat semua kontak
- ✅ Cari kontak
- ✅ Edit kontak
- ✅ Hapus kontak
- ✅ Simpan ke file JSON
- ✅ Muat dari file JSON saat startup

## Implementasi

```python
import json
import os

class Kontak:
    def __init__(self, nama, telepon, email="", catatan=""):
        self.nama = nama
        self.telepon = telepon
        self.email = email
        self.catatan = catatan
    
    def to_dict(self):
        return {
            "nama": self.nama,
            "telepon": self.telepon,
            "email": self.email,
            "catatan": self.catatan
        }
    
    @classmethod
    def from_dict(cls, data):
        return cls(
            data["nama"],
            data["telepon"],
            data.get("email", ""),
            data.get("catatan", "")
        )
    
    def __str__(self):
        info = f"📱 {self.nama} | ☎️ {self.telepon}"
        if self.email:
            info += f" | ✉️ {self.email}"
        if self.catatan:
            info += f" | 📝 {self.catatan}"
        return info


class BukuTelepon:
    FILE = "kontak.json"
    
    def __init__(self):
        self.kontak_list = []
        self.muat()
    
    def muat(self):
        if os.path.exists(self.FILE):
            try:
                with open(self.FILE, "r", encoding="utf-8") as f:
                    data = json.load(f)
                    self.kontak_list = [Kontak.from_dict(k) for k in data]
                print(f"✅ Memuat {len(self.kontak_list)} kontak dari file.")
            except Exception as e:
                print(f"⚠️ Gagal memuat file: {e}")
    
    def simpan(self):
        try:
            with open(self.FILE, "w", encoding="utf-8") as f:
                json.dump([k.to_dict() for k in self.kontak_list], f, 
                          indent=2, ensure_ascii=False)
            print(f"💾 Tersimpan! ({len(self.kontak_list)} kontak)")
        except Exception as e:
            print(f"❌ Gagal menyimpan: {e}")
    
    def tambah(self, kontak):
        self.kontak_list.append(kontak)
        self.simpan()
    
    def cari(self, query):
        q = query.lower()
        return [k for k in self.kontak_list 
                if q in k.nama.lower() or q in k.telepon]
    
    def hapus(self, nama):
        asli = len(self.kontak_list)
        self.kontak_list = [k for k in self.kontak_list 
                            if k.nama.lower() != nama.lower()]
        if len(self.kontak_list) < asli:
            self.simpan()
            return True
        return False
    
    def tampilkan_semua(self):
        if not self.kontak_list:
            print("📭 Buku telepon kosong.")
            return
        
        print(f"\n📒 BUKU TELEPON ({len(self.kontak_list)} kontak)")
        print("─" * 50)
        for i, k in enumerate(sorted(self.kontak_list, key=lambda x: x.nama), 1):
            print(f"{i:>3}. {k}")
        print("─" * 50)


def main():
    buku = BukuTelepon()
    
    while True:
        print("\n╔══════════════════════════╗")
        print("║   MANAJEMEN KONTAK 📒    ║")
        print("╠══════════════════════════╣")
        print("║ 1. Lihat semua kontak    ║")
        print("║ 2. Tambah kontak         ║")
        print("║ 3. Cari kontak           ║")
        print("║ 4. Hapus kontak          ║")
        print("║ 5. Keluar                ║")
        print("╚══════════════════════════╝")
        
        pilihan = input("\nPilihan: ").strip()
        
        if pilihan == "1":
            buku.tampilkan_semua()
        
        elif pilihan == "2":
            print("\n📝 TAMBAH KONTAK BARU")
            nama = input("Nama: ").strip()
            if not nama:
                print("Nama tidak boleh kosong!")
                continue
            telepon = input("No. Telepon: ").strip()
            email = input("Email (kosongkan jika tidak ada): ").strip()
            catatan = input("Catatan (opsional): ").strip()
            
            kontak_baru = Kontak(nama, telepon, email, catatan)
            buku.tambah(kontak_baru)
            print(f"✅ Kontak '{nama}' berhasil ditambahkan!")
        
        elif pilihan == "3":
            query = input("Cari nama/telepon: ").strip()
            hasil = buku.cari(query)
            
            if hasil:
                print(f"\n🔍 Ditemukan {len(hasil)} kontak:")
                for k in hasil:
                    print(f"  • {k}")
            else:
                print(f"Tidak ada kontak yang cocok dengan '{query}'")
        
        elif pilihan == "4":
            nama = input("Nama kontak yang dihapus: ").strip()
            if buku.hapus(nama):
                print(f"✅ Kontak '{nama}' berhasil dihapus.")
            else:
                print(f"❌ Kontak '{nama}' tidak ditemukan.")
        
        elif pilihan == "5":
            print("\n👋 Sampai jumpa!")
            break
        
        else:
            print("❌ Pilihan tidak valid!")

main()
```
